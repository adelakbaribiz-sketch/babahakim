    47→  const load = useCallback(async () => {

   291→function ProgressTrack({ status }: { status: string }) {
   292→  const { t } = useApp()
   293→  const flow = ['WAITING_FOR_EXPERT_1', 'EXPERT_1_REVIEW', 'WAITING_FOR_USER_APPROVAL', 'WAITING_FOR_EXPERT_1_PRESCRIPTION', 'WAITING_FOR_EXPERT_2', 'WAITING_FOR_DOCTOR', 'PRESCRIPTION_APPROVED', 'PROFORMA_CREATED', 'WAITING_FOR_PAYMENT', 'PAYMENT_CONFIRMED', 'SHIPPED', 'DELIVERED']
   294→  const idx = flow.indexOf(status)
   295→  if (idx === -1) return null
   296→  return (
   297→    <div className="flex items-center gap-1 overflow-x-auto pb-1">
   298→      {flow.map((s, i) => (
   299→        <React.Fragment key={s}>
   300→          <div className={`shrink-0 rounded-full px-2.5 py-1 text-[10px] whitespace-nowrap ${i < idx ? 'bg-status-info text-status-info-fg font-medium' : i === idx ? 'bg-primary text-primary-foreground' : 'bg-muted text-muted-foreground'}`}>
   301→            {(STATUS_META as Record<string, { fa: string; en: string }>)[s]?.fa || s}
   302→          </div>
   303→          {i < flow.length - 1 && <div className={`h-0.5 w-4 shrink-0 ${i < idx ? 'bg-brand-light' : 'bg-muted'}`} />}
   304→        </React.Fragment>
   305→      ))}
   306→    </div>
   307→  )
   308→}

   330→interface RxItemDraft { productId?: string; drugName: string; dosage: string; frequency: string; duration: string; instructions: string; warnings: string }
   331→export function PrescriptionEditor({ onSubmit, products }: { onSubmit: (items: RxItemDraft[], notes: string) => Promise<boolean>; products?: { id: string; nameFa: string }[] }) {
   332→  const { t, lang } = useApp()
   333→  const [items, setItems] = useState<RxItemDraft[]>([{ drugName: '', dosage: '', frequency: '', duration: '', instructions: '', warnings: '' }])
   334→  const [notes, setNotes] = useState('')
   335→  const [busy, setBusy] = useState(false)
   336→
   337→  const add = () => setItems([...items, { drugName: '', dosage: '', frequency: '', duration: '', instructions: '', warnings: '' }])
   338→  const set = (i: number, k: keyof RxItemDraft, v: string) => setItems(items.map((it, idx) => idx === i ? { ...it, [k]: v } : it))
   339→
   340→  return (
   341→    <Card className="border-primary/30">
   342→      <CardHeader className="pb-3"><CardTitle className="text-base">{t('prescription')} — {lang === 'fa' ? 'تنظیم نسخه' : 'Compose'}</CardTitle></CardHeader>
   343→      <CardContent className="space-y-4">
   344→        {items.map((it, i) => (
   345→          <div key={i} className="rounded-lg border p-3 space-y-2.5 relative">
   346→            {items.length > 1 && <Button variant="ghost" size="sm" className="absolute top-2 end-2 h-6 text-destructive" onClick={() => setItems(items.filter((_, x) => x !== i))}>×</Button>}
   347→            <div className="grid sm:grid-cols-2 gap-2.5">
   348→              <div className="space-y-1"><Label className="text-xs">{lang === 'fa' ? 'نام دارو / محصول' : 'Drug'} *</Label>
   349→                {products && products.length > 0 ? (
   350→                  <Select onValueChange={(v) => { const p = products.find((x) => x.id === v); set(i, 'drugName', p?.nameFa || ''); setItems(items.map((x, idx) => idx === i ? { ...x, productId: v } : x)) }}>
   351→                    <SelectTrigger><SelectValue placeholder={lang === 'fa' ? 'انتخاب یا تایپ' : 'Select or type'} /></SelectTrigger>
   352→                    <SelectContent>{products.map((p) => <SelectItem key={p.id} value={p.id}>{p.nameFa}</SelectItem>)}</SelectContent>
   353→                  </Select>

   443→function OperatorShipping({ caseId, status, onSubmit }: { caseId: string; status: string; onSubmit: () => void }) {
   444→  const { t, lang, api } = useApp()
   445→  const [open, setOpen] = useState(false)
   446→  const [busy, setBusy] = useState(false)
   447→  const [form, setForm] = useState({ method: 'MOTORCYCLE', driverName: '', driverPhone: '', trackingCode: '', cost: '50000' })
   448→
   449→  async function ship() {
   450→    setBusy(true)
   451→    try {
   452→      await api(`/api/v1/cases/${caseId}/action`, { method: 'POST', json: { action: 'START_SHIPPING', ...form, cost: Number(form.cost) } })

   454→      setOpen(false)
   455→      onSubmit()
